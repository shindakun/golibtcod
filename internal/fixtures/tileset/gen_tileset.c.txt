#include <stdio.h>
#include <stdlib.h>

// Same body as sys_c.c's TCOD_load_binary_file_, inlined so the harness
// does not have to link sys_c.c (which pulls in SDL).
unsigned char* TCOD_load_binary_file_(const char* path, size_t* size) {
  FILE* file = fopen(path, "rb");
  if (!file) return NULL;
  fseek(file, 0, SEEK_END);
  size_t fsize = (size_t)ftell(file);
  fseek(file, 0, SEEK_SET);
  unsigned char* buffer = malloc(fsize);
  if (buffer && fread(buffer, 1, fsize, file) != fsize) { free(buffer); buffer = NULL; }
  fclose(file);
  if (size) *size = fsize;
  return buffer;
}
#include "tileset.h"
#include "tileset_bdf.h"

int main(int argc, char** argv) {
  if (argc < 2) return 1;
  TCOD_Tileset* ts = TCOD_load_bdf(argv[1]);
  if (!ts) { fprintf(stderr, "load failed\n"); return 1; }
  printf("== bdf %s\n", argv[1]);
  printf("cell %d %d\n", TCOD_tileset_get_tile_width_(ts), TCOD_tileset_get_tile_height_(ts));
  int w = TCOD_tileset_get_tile_width_(ts), h = TCOD_tileset_get_tile_height_(ts);
  TCOD_ColorRGBA* buf = malloc(sizeof(TCOD_ColorRGBA) * w * h);
  // Dump alpha coverage for a fixed codepoint sweep.
  int cps[] = {32,33,48,49,65,66,90,97,98,122,169,177,948,9608,9617};
  for (unsigned i = 0; i < sizeof(cps)/sizeof(cps[0]); i++) {
    int cp = cps[i];
    if (TCOD_tileset_get_tile_(ts, cp, buf) < 0) { printf("cp %d MISSING\n", cp); continue; }
    printf("cp %d\n", cp);
    for (int y = 0; y < h; y++) {
      for (int x = 0; x < w; x++) printf("%d", buf[y*w+x].a >= 128 ? 1 : 0);
      printf("\n");
    }
  }
  free(buf);
  TCOD_tileset_delete(ts);
  return 0;
}
